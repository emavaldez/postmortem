Para cada registro recorre hacia atrás las filas del mismo cliente dentro de la ventana, descarta valores no finitos y acumula sumas usando como x la distancia en meses al mes actual; con al menos 2 valores devuelve la pendiente por mínimos cuadrados (tendencia) y el promedio, y NA en caso contrario.

288Rcpp::List fhist_tend_avg(
289 Rcpp::NumericVector valores,
290 Rcpp::IntegerVector meses,
291 Rcpp::IntegerVector grupos,
292 int ventana) {
293
294 int n = valores.size();
295
296 Rcpp::NumericVector tendencia(n, NA_REAL);
297 Rcpp::NumericVector promedio(n, NA_REAL);
298
299 for (int i = 0; i < n; i++) {
300
301 if (i % 10000 == 0) Rcpp::checkUserInterrupt();
302
303 int cantidad = 0;
304 double sx = 0.0;
305 double sy = 0.0;
306 double sxx = 0.0;
307 double sxy = 0.0;
308
309 int mes_desde = meses[i] - ventana + 1;
310
311 for (int j = i; j >= 0; j--) {
312
313 if (grupos[j] != grupos[i]) break;
314 if (meses[j] < mes_desde) break;
315
316 double y = valores[j];
317
318 // Excluir NA, NaN e infinitos
319 if (!R_FINITE(y)) continue;
320
321 // Distancia mensual respecto al mes actual
322 double x = meses[j] - meses[i];
323
324 cantidad++;
325 sx += x;
326 sy += y;
327 sxx += x * x;
328 sxy += x * y;
329 }
330
331 if (cantidad >= 2) {
332
333 double denominador =
334 cantidad * sxx - sx * sx;
335
336 if (denominador > 0.0) {
337 tendencia[i] =
338 (cantidad * sxy - sx * sy) / denominador;
339 }
340
341 promedio[i] = sy / cantidad;
342 }
343 }
344
345 return Rcpp::List::create(
346 Rcpp::Named("tendencia") = tendencia,
347 Rcpp::Named("promedio") = promedio
