# [Anotación · líneas 47–66] `positive_cluster_vector` asigna a cada fila BAJA+2 el grupo de su cliente (0
#   para las demás) y se detiene si algún BAJA+2 no tiene grupo. `base_weight_vector` traduce el peso por mes a
#   un peso por fila.
def positive_cluster_vector(ids,y,cluster_by_id: dict[int,int]) -> np.ndarray:
    ids=np.asarray(ids,dtype=np.int64)
    y=np.asarray(y,dtype=np.int8)
    out=np.zeros(len(ids),dtype=np.int8)
    pos=np.flatnonzero(y==1)
    missing=[]
    for i in pos:
        k=cluster_by_id.get(int(ids[i]))
        if k is None:
            missing.append(int(ids[i]))
        else:
            out[i]=int(k)
    A.require(not missing,f"BAJA+2 sin cluster: {missing[:10]} (n={len(missing)})")
    A.require(np.all(np.isin(out[pos],A.CLUSTERS)),"Cluster positivo fuera de 1..5")
    return out


def base_weight_vector(months: np.ndarray,weights_by_month: dict[int,float]) -> np.ndarray:
    months=np.asarray(months,dtype=np.int64)
    return np.asarray([float(weights_by_month.get(int(m),0.0)) for m in months],dtype=np.float64)


# [Anotación · líneas 69–76] `load_cluster_map` lee el CSV de grupos y exige las columnas `numero_de_cliente`
#   y `cluster_ref`, sin clientes repetidos y con los grupos 1 a 5 presentes.
def load_cluster_map(path: Path) -> tuple[pd.DataFrame,dict[int,int]]:
    z=pd.read_csv(path)
    A.require({"numero_de_cliente","cluster_ref"}<=set(z.columns),"CSV clusters incompleto")
    A.require(not z.numero_de_cliente.duplicated().any(),"CSV clusters con IDs duplicados")
    z["numero_de_cliente"]=z.numero_de_cliente.astype("int64")
    z["cluster_ref"]=z.cluster_ref.astype("int8")
    A.require(set(z.cluster_ref.unique())==set(A.CLUSTERS),"Clusters esperados 1..5 no presentes")
    return z,{int(r.numero_de_cliente):int(r.cluster_ref) for r in z.itertuples(index=False)}

# [Anotación · líneas 102–115] `load_sf_control` lee los puntajes de 202106 del candidato de la etapa 4 de
#   SUPERFINAL para las semillas pedidas y valida que clientes y clase coincidan.
def load_sf_control(sf: Path,seeds: list[int],ids: np.ndarray,y: np.ndarray) -> dict[int,np.ndarray]:
    out={}
    root=sf/"stage4_confirm20/candidate"
    for seed in seeds:
        cp=root/f"seed_{seed}.npz"
        A.require(cp.is_file(),f"Falta control SF seed={seed}")
        with np.load(cp,allow_pickle=False) as z:
            zid=np.asarray(z["ids"],dtype=np.int64)
            zy=np.asarray(z["y"],dtype=np.int8)
            sc=np.asarray(z["scores"],dtype=np.float64)
        A.require(np.array_equal(zid,ids) and np.array_equal(zy,y),f"Control SF desalineado seed={seed}")
        A.require(np.isfinite(sc).all(),f"Control SF no finito seed={seed}")
        out[seed]=sc.copy()
    return out


# [Anotación · líneas 118–189] `train_scores` calcula los pesos por fila del brazo con `make_weights`,
#   descarta filas de peso 0 y, para cada semilla, reutiliza el `.npz`, copia un puntaje heredado o entrena un
#   LightGBM con dataset propio de la semilla (parámetros y rondas de SUPERFINAL); guarda puntajes, brazo,
#   semilla y el detalle de pesos.
def train_scores(
    data: dict,
    train_ids: np.ndarray,
    pc_train: np.ndarray,
    arm: str,
    base_weights: np.ndarray,
    features: list[str],
    params: dict,
    rounds: int,
    seeds: list[int],
    out_dir: Path,
    identity: str,
    reuse_scores: dict[int,np.ndarray]|None=None,
) -> tuple[dict[int,np.ndarray],dict]:
    import lightgbm as lgb

    reuse_scores=reuse_scores or {}
    weights,weight_detail=A.make_weights(data["y_train"],pc_train,base_weights,arm)
    keep=weights>0
    A.require(keep.any(),"Pesos dejaron train vacio")
    x=(
        data["x_train"] if bool(np.all(keep))
        else data["x_train"].loc[keep].reset_index(drop=True)
    )
    y=np.asarray(data["y_train"],dtype=np.int8)[keep]
    w=weights[keep]
    ids_eval=np.asarray(data["ids"],dtype=np.int64)
    y_eval=np.asarray(data["y_eval"],dtype=np.int8)

    out_dir.mkdir(parents=True,exist_ok=True)
    result={}
    for i,seed in enumerate(seeds,1):
        cp=out_dir/f"seed_{seed}.npz"
        if cp.is_file():
            with np.load(cp,allow_pickle=False) as z:
                A.require(str(z["identity"].item())==identity,f"CP identity seed={seed}")
                A.require(str(z["arm"].item())==arm,f"CP arm seed={seed}")
                A.require(int(z["seed"].item())==seed,f"CP seed interno {seed}")
                A.require(np.array_equal(z["ids"],ids_eval) and np.array_equal(z["y"],y_eval),
                          f"CP keys seed={seed}")
                sc=np.asarray(z["scores"],dtype=np.float64)
            A.require(np.isfinite(sc).all(),"CP scores no finitos")
            result[seed]=sc.copy()
            print(f"REUSAR {arm} {i}/{len(seeds)} seed={seed}",flush=True)
            continue

        if seed in reuse_scores:
            sc=np.asarray(reuse_scores[seed],dtype=np.float64)
            A.require(sc.shape==ids_eval.shape and np.isfinite(sc).all(),"Reuse score invalido")
            reused=True
            print(f"COPIAR {arm} {i}/{len(seeds)} seed={seed}",flush=True)
        else:
            print(f"ENTRENAR {arm} {i}/{len(seeds)} seed={seed}",flush=True)
            ds,seed_params=V2.dataset_for_seed(x,y,w,features,params,seed)
            model=lgb.train(seed_params,ds,num_boost_round=int(rounds))
            sc=np.asarray(model.predict(data["x_eval"],num_threads=C.THREADS),dtype=np.float64)
            A.require(sc.shape==ids_eval.shape and np.isfinite(sc).all(),"Scores nuevos invalidos")
            del model,ds
            gc.collect()
            reused=False

        tmp=cp.with_suffix(".tmp")
        with tmp.open("wb") as fh:
            np.savez_compressed(
                fh,identity=np.array(identity),arm=np.array(arm),seed=np.array(seed),
                ids=ids_eval,y=y_eval,scores=sc,reused=np.array(reused),
                weight_detail=np.array(json.dumps(weight_detail,sort_keys=True))
            )
        tmp.replace(cp)
        result[seed]=sc.copy()

    return result,weight_detail


# [Anotación · líneas 192–230] `evaluate` calcula las métricas en el corte 13.500 por semilla y para el
#   promedio de semillas. `comparison_row` arma la fila de comparación contra el control: diferencias de
#   ganancia, de BAJA+2 incluidos y de incluidos por grupo, resumen pareado, y `passed` si la ganancia y los
#   incluidos de los grupos 2 y 5 aumentan, la mediana pareada es positiva y se alcanza el mínimo de semillas a
#   favor.
def evaluate(ids,y,pc_eval,score_by_seed,seeds) -> tuple[dict,list[dict],np.ndarray]:
    rows=[]
    for seed in seeds:
        m=A.topn_metrics(ids,y,score_by_seed[seed],pc_eval,A.N_FIXED)
        rows.append({"seed":seed,**m})
    ens=np.vstack([score_by_seed[s] for s in seeds]).mean(axis=0,dtype=np.float64)
    em=A.topn_metrics(ids,y,ens,pc_eval,A.N_FIXED)
    return em,rows,ens


def comparison_row(arm,cand,ctrl,cand_seed_rows,ctrl_seed_rows,min_wins) -> dict:
    pair=A.paired_summary(
        [r["gain"] for r in cand_seed_rows],
        [r["gain"] for r in ctrl_seed_rows],
    )
    row={
        "arm":arm,
        "ensemble_gain":int(cand["gain"]),
        "control_gain":int(ctrl["gain"]),
        "delta_gain":int(cand["gain"]-ctrl["gain"]),
        "ensemble_hits":int(cand["hits"]),
        "control_hits":int(ctrl["hits"]),
        "delta_hits":int(cand["hits"]-ctrl["hits"]),
        "ensemble_c25_captured":int(cand["c25_captured"]),
        "control_c25_captured":int(ctrl["c25_captured"]),
        "delta_c25_captured":int(cand["c25_captured"]-ctrl["c25_captured"]),
        "paired_mean":pair["mean"],"paired_median":pair["median"],
        "paired_sd":pair["sd"],"paired_wins":pair["wins"],
        "paired_ties":pair["ties"],"paired_losses":pair["losses"],
    }
    for k in A.CLUSTERS:
        row[f"c{k}_captured"]=int(cand[f"c{k}_captured"])
        row[f"control_c{k}_captured"]=int(ctrl[f"c{k}_captured"])
        row[f"delta_c{k}_captured"]=int(cand[f"c{k}_captured"]-ctrl[f"c{k}_captured"])
    row["passed"]=bool(
        arm!="CONTROL" and row["delta_gain"]>0 and row["delta_c25_captured"]>0
        and row["paired_median"]>0 and row["paired_wins"]>=min_wins
    )
    return row


# [Anotación · líneas 233–272] `preflight` registra los hashes del CSV de grupos, del plan y del código (y, si
#   ya hay un preflight guardado, exige que no hayan cambiado). Verifica que haya 4.067 clientes BAJA+2 en
#   202103–202106 y que el CSV de grupos contenga exactamente esos clientes; guarda tamaños de grupo, cantidad
#   de variables, parámetros, rondas, política y corte fijo.
def preflight(home: Path,cluster_path: Path) -> dict:
    out=home/RUN/"stage0_preflight"
    rp=out/"RESULT.json"
    A.require(cluster_path.is_file(),f"Falta CSV clusters: {cluster_path}")
    current_inputs={
        "cluster_sha256":A.digest(cluster_path),
        "plan_sha256":A.digest(HERE/"plan.json"),
        "runner_sha256":CODE_SHA,
        "common_sha256":A.digest(HERE/"cluster_aware_common.py"),
    }
    if rp.is_file():
        old=A.load_json(rp)
        A.require(old.get("status")=="PREFLIGHT_OK","Preflight cacheado inválido")
        A.require(old.get("source_identity")==SOURCE_IDENTITY,"Fuente cambió desde preflight")
        for key,value in current_inputs.items():
            A.require(old.get(key)==value,f"Input cambió después del preflight: {key}")
        return old
    src=source(home)
    clusters,cmap=load_cluster_map(cluster_path)
    labels=pd.read_parquet(home/"c1m1femasiva03_s3/resultados_s1/fe_bank_dev/etiquetas.parquet")
    mature=labels[labels.foto_mes.isin([202103,202104,202105,202106])]
    b2=mature[mature.clase_ternaria.eq("BAJA+2")]
    A.require(len(b2)==4067 and b2.numero_de_cliente.nunique()==4067,"Cohorte BAJA+2 madura inesperada")
    got=set(b2.numero_de_cliente.astype("int64"))
    have=set(clusters.numero_de_cliente.astype("int64"))
    A.require(got==have,"reference_labels no coincide exactamente con BAJA+2 marzo-junio")
    counts=clusters.cluster_ref.value_counts().sort_index().to_dict()
    result={
        "status":"PREFLIGHT_OK","run":RUN,"source_identity":SOURCE_IDENTITY,
        "cluster_file":str(cluster_path),**current_inputs,
        "n_cluster_labels":int(len(clusters)),
        "cluster_counts":{str(int(k)):int(v) for k,v in counts.items()},
        "features_sha256":A.fingerprint(src["features"]),
        "n_features":len(src["features"]),
        "params":src["params"],"rounds":src["rounds"],"policy":src["policy"],
        "fixed_n":A.N_FIXED,
        "methodological_status":"DIAGNOSTIC_INFORMED_COMPETITIVE_NOT_INDEPENDENT_TEMPORAL_VALIDATION",
    }
    A.write_json(rp,result)
    return result


# [Anotación · líneas 275–332] `stage1_screen` (prueba de mayo): entrena con 202103 y evalúa en 202105 los
#   cuatro brazos (CONTROL, C25_FOCUSED, DIFFICULTY_JUNE y BALANCED_CAPPED) con 3 semillas y peso base 1.
#   Compara cada brazo con CONTROL con un mínimo de 2 semillas a favor y elige, entre los que pasan, el de mayor
#   ganancia del promedio de semillas. Escribe `comparison.csv`, `seed_metrics.csv`, `weight_details.json` y
#   `RESULT.json`.
def stage1_screen(home: Path,cluster_path: Path) -> dict:
    root=home/RUN/"stage1_screen"
    rp=root/"RESULT.json"
    if rp.is_file():
        return A.load_json(rp)
    src=source(home)
    ctx=C.preflight_base(home)
    _,cmap=load_cluster_map(cluster_path)
    data=C.load_fold(ctx,src["features"],[202103],202105,Path(src["s0"]["eval_bank"]))
    labels=pd.read_parquet(ctx["paths"]["bank_orig"]/"etiquetas.parquet")
    trlab=labels[labels.foto_mes.eq(202103)].reset_index(drop=True)
    A.require(len(trlab)==len(data["y_train"]),"Screen train labels desalineadas")
    train_ids=trlab.numero_de_cliente.to_numpy(np.int64)
    pc_train=positive_cluster_vector(train_ids,data["y_train"],cmap)
    pc_eval=positive_cluster_vector(data["ids"],data["y_eval"],cmap)
    base=np.ones(len(train_ids),dtype=np.float64)

    stage_identity=A.fingerprint({
        "run":RUN,"stage":1,"plan":PLAN,"source":SOURCE_IDENTITY,"code_sha":CODE_SHA,
        "features":A.fingerprint(src["features"]),"params":src["params"],"rounds":src["rounds"],
        "train":[202103],"eval":202105,"seeds":SCREEN_SEEDS,
    })
    all_scores={}
    weight_details={}
    metrics={}
    seedrows={}
    for arm in A.ARMS:
        aid=A.fingerprint({"parent":stage_identity,"arm":arm})
        sc,wd=train_scores(
            data,train_ids,pc_train,arm,base,src["features"],src["params"],src["rounds"],
            SCREEN_SEEDS,root/"scores"/arm,aid
        )
        em,sr,_=evaluate(data["ids"],data["y_eval"],pc_eval,sc,SCREEN_SEEDS)
        all_scores[arm]=sc; weight_details[arm]=wd; metrics[arm]=em; seedrows[arm]=sr

    control=metrics["CONTROL"]; cseed=seedrows["CONTROL"]
    rows=[]
    for arm in A.ARMS:
        if arm=="CONTROL":
            row=comparison_row("CONTROL",control,control,cseed,cseed,2)
            row["passed"]=True
        else:
            row=comparison_row(arm,metrics[arm],control,seedrows[arm],cseed,2)
        rows.append(row)
    winner=A.choose_screen_winner(rows)
    frame=pd.DataFrame(rows)
    frame.to_csv(root/"comparison.csv",index=False)
    pd.DataFrame([
        {"arm":arm,**r} for arm in A.ARMS for r in seedrows[arm]
    ]).to_csv(root/"seed_metrics.csv",index=False)
    A.write_json(root/"weight_details.json",weight_details)
    result={
        "status":"SCREEN_PASS" if winner else "STOP_NO_SCREEN_WINNER",
        "identity":stage_identity,"winner":winner,"fixed_n":A.N_FIXED,
        "seeds":SCREEN_SEEDS,"rows":rows,"weight_details":weight_details,
    }
    A.write_json(rp,result)
    return result


# [Anotación · líneas 335–394] `stage2_confirm5`: con 202103–202104 y evaluación en 202106, primero entrena
#   CONTROL con la semilla 808 y exige puntajes idénticos a los de SUPERFINAL. Después entrena el brazo ganador
#   con 5 semillas y lo compara con los puntajes de SUPERFINAL de esas semillas (mínimo 3 a favor). Si la prueba
#   de mayo no tuvo ganador, no entrena.
def stage2_confirm5(home: Path,cluster_path: Path,s1: dict) -> dict:
    root=home/RUN/"stage2_confirm5"
    rp=root/"RESULT.json"
    if rp.is_file():
        return A.load_json(rp)
    if s1.get("winner") is None:
        r={"status":"SKIPPED_SCREEN_FAIL","winner":None}
        A.write_json(rp,r); return r

    src=source(home); ctx=C.preflight_base(home); _,cmap=load_cluster_map(cluster_path)
    data=C.load_fold(ctx,src["features"],[202103,202104],202106,Path(src["s0"]["eval_bank"]))
    labels=pd.read_parquet(ctx["paths"]["bank_orig"]/"etiquetas.parquet")
    trlab=labels[labels.foto_mes.isin([202103,202104])].reset_index(drop=True)
    A.require(len(trlab)==len(data["y_train"]),"Confirm5 train labels desalineadas")
    train_ids=trlab.numero_de_cliente.to_numpy(np.int64)
    pc_train=positive_cluster_vector(train_ids,data["y_train"],cmap)
    pc_eval=positive_cluster_vector(data["ids"],data["y_eval"],cmap)
    base=base_weight_vector(data["train_months"],V1.RECENCY[src["policy"]])
    winner=str(s1["winner"])

    # Canary exacto: CONTROL seed 808 por el nuevo camino de pesos.
    cid=A.fingerprint({
        "run":RUN,"stage":"canary","source":SOURCE_IDENTITY,"seed":808,"code_sha":CODE_SHA,
        "features":A.fingerprint(src["features"]),"params":src["params"],
        "rounds":src["rounds"],"policy":src["policy"],
    })
    csc,_=train_scores(
        data,train_ids,pc_train,"CONTROL",base,src["features"],src["params"],src["rounds"],
        [808],root/"canary_control",cid
    )
    sf808=load_sf_control(src["root"],[808],data["ids"],data["y_eval"])[808]
    max_abs=float(np.max(np.abs(csc[808]-sf808)))
    exact=bool(np.array_equal(csc[808],sf808))
    A.require(exact and max_abs==0.0,f"CANARY CONTROL NO EXACTO max_abs={max_abs}")
    canary={"status":"CANARY_EXACT_OK","seed":808,"array_equal":exact,"max_abs":max_abs}
    A.write_json(root/"CANARY.json",canary)

    identity=A.fingerprint({
        "run":RUN,"stage":2,"plan":PLAN,"screen_identity":s1["identity"],"code_sha":CODE_SHA,
        "winner":winner,"source":SOURCE_IDENTITY,"seeds":CONFIRM5_SEEDS,
    })
    candidate,wd=train_scores(
        data,train_ids,pc_train,winner,base,src["features"],src["params"],src["rounds"],
        CONFIRM5_SEEDS,root/"candidate",identity
    )
    control=load_sf_control(src["root"],CONFIRM5_SEEDS,data["ids"],data["y_eval"])
    cm,csr,_=evaluate(data["ids"],data["y_eval"],pc_eval,candidate,CONFIRM5_SEEDS)
    bm,bsr,_=evaluate(data["ids"],data["y_eval"],pc_eval,control,CONFIRM5_SEEDS)
    row=comparison_row(winner,cm,bm,csr,bsr,3)
    pd.DataFrame([row]).to_csv(root/"comparison.csv",index=False)
    pd.DataFrame([{"kind":"candidate",**r} for r in csr]+[{"kind":"control",**r} for r in bsr]).to_csv(
        root/"seed_metrics.csv",index=False
    )
    result={
        "status":"CONFIRM5_PASS" if row["passed"] else "STOP_CONFIRM5_FAIL",
        "identity":identity,"winner":winner,"canary":canary,"comparison":row,
        "weight_detail":wd,"fixed_n":A.N_FIXED,"seeds":CONFIRM5_SEEDS,
    }
    A.write_json(rp,result)
    return result


# [Anotación · líneas 397–408] `read_candidate_scores` lee los puntajes del candidato de la etapa anterior
#   validando identidad, brazo y claves.
def read_candidate_scores(root: Path,identity: str,arm: str,seeds: list[int],ids,y) -> dict[int,np.ndarray]:
    out={}
    for seed in seeds:
        cp=root/f"seed_{seed}.npz"
        A.require(cp.is_file(),f"Falta candidato previo {cp}")
        with np.load(cp,allow_pickle=False) as z:
            A.require(str(z["identity"].item())==identity and str(z["arm"].item())==arm,
                      f"Candidato previo metadata seed={seed}")
            A.require(np.array_equal(z["ids"],ids) and np.array_equal(z["y"],y),
                      f"Candidato previo keys seed={seed}")
            out[seed]=np.asarray(z["scores"],dtype=np.float64).copy()
    return out


# [Anotación · líneas 411–462] `stage3_confirm20`: entrena el brazo ganador con las 20 semillas (reutiliza las
#   5 anteriores) y compara en 202106 contra los 20 puntajes de SUPERFINAL. Pasa si el promedio de semillas
#   incluye al menos 2 BAJA+2 más, al menos 1 más de los grupos 2 y 5, la diferencia pareada media y mediana son
#   positivas y gana en al menos 14 semillas.
def stage3_confirm20(home: Path,cluster_path: Path,s1: dict,s2: dict) -> dict:
    root=home/RUN/"stage3_confirm20"
    rp=root/"RESULT.json"
    if rp.is_file():
        return A.load_json(rp)
    if s2.get("status")!="CONFIRM5_PASS":
        r={"status":"SKIPPED_CONFIRM5_FAIL","winner":s1.get("winner")}
        A.write_json(rp,r); return r

    src=source(home); ctx=C.preflight_base(home); _,cmap=load_cluster_map(cluster_path)
    data=C.load_fold(ctx,src["features"],[202103,202104],202106,Path(src["s0"]["eval_bank"]))
    labels=pd.read_parquet(ctx["paths"]["bank_orig"]/"etiquetas.parquet")
    trlab=labels[labels.foto_mes.isin([202103,202104])].reset_index(drop=True)
    train_ids=trlab.numero_de_cliente.to_numpy(np.int64)
    A.require(len(train_ids)==len(data["y_train"]),"Confirm20 train labels desalineadas")
    pc_train=positive_cluster_vector(train_ids,data["y_train"],cmap)
    pc_eval=positive_cluster_vector(data["ids"],data["y_eval"],cmap)
    base=base_weight_vector(data["train_months"],V1.RECENCY[src["policy"]])
    winner=str(s1["winner"])

    reuse=read_candidate_scores(
        home/RUN/"stage2_confirm5/candidate",s2["identity"],winner,
        CONFIRM5_SEEDS,data["ids"],data["y_eval"]
    )
    identity=A.fingerprint({
        "run":RUN,"stage":3,"plan":PLAN,"confirm5_identity":s2["identity"],"code_sha":CODE_SHA,
        "winner":winner,"source":SOURCE_IDENTITY,"seeds":SEEDS20,
    })
    candidate,wd=train_scores(
        data,train_ids,pc_train,winner,base,src["features"],src["params"],src["rounds"],
        SEEDS20,root/"candidate",identity,reuse_scores=reuse
    )
    control=load_sf_control(src["root"],SEEDS20,data["ids"],data["y_eval"])
    cm,csr,_=evaluate(data["ids"],data["y_eval"],pc_eval,candidate,SEEDS20)
    bm,bsr,_=evaluate(data["ids"],data["y_eval"],pc_eval,control,SEEDS20)
    row=comparison_row(winner,cm,bm,csr,bsr,14)
    pair=A.paired_summary([r["gain"] for r in csr],[r["gain"] for r in bsr])
    row["passed"]=bool(
        row["delta_hits"]>=2 and row["delta_c25_captured"]>=1
        and pair["mean"]>0 and pair["median"]>0 and pair["wins"]>=14
    )

    pd.DataFrame([{"kind":"candidate",**r} for r in csr]+[{"kind":"control",**r} for r in bsr]).to_csv(
        root/"seed_metrics.csv",index=False
    )
    result={
        "status":"CONFIRM20_PASS" if row["passed"] else "STOP_CONFIRM20_FAIL",
        "identity":identity,"winner":winner,"comparison":row,"paired":pair,
        "weight_detail":wd,"fixed_n":A.N_FIXED,"seeds":SEEDS20,
    }
    A.write_json(rp,result)
    return result


# [Anotación · líneas 465–509] `stage4_august` (sólo si la etapa anterior pasó): carga del banco final de
#   SUPERFINAL las filas de 202103–202106 con las variables elegidas, arma la clase binaria BAJA+2, asigna grupo
#   a los positivos, aplica el peso por mes de la política y calcula los pesos por fila del brazo ganador. Carga
#   las 164.647 filas de 202108 y calcula la identidad de la etapa.
def stage4_august(home: Path,cluster_path: Path,s1: dict,s3: dict) -> dict:
    import lightgbm as lgb

    root=home/RUN/"stage4_august"
    rp=root/"RESULT.json"
    if rp.is_file():
        return A.load_json(rp)
    if s3.get("status")!="CONFIRM20_PASS":
        r={"status":"SKIPPED_CONFIRM20_FAIL","winner":s1.get("winner"),"no_submit":True}
        A.write_json(rp,r); return r

    src=source(home); ctx=C.preflight_base(home); _,cmap=load_cluster_map(cluster_path)
    features=src["features"]; winner=str(s1["winner"])
    bank=src["root"]/"stage5_final/bank"
    A.require((bank/"manifest.json").is_file(),"Falta banco final SUPERFINAL")
    C.add_fe_path(ctx["paths"])
    from cargar_bloques import cargar_bloques
    derived=C.split_derived(features,ctx)

    trdf,_=cargar_bloques(bank,meses=[202103,202104,202105,202106],incluir_raw=True,features=derived)
    labels=pd.read_parquet(ctx["paths"]["bank_orig"]/"etiquetas.parquet")
    trlab=labels[labels.foto_mes.isin([202103,202104,202105,202106])].reset_index(drop=True)
    A.require(trdf[["numero_de_cliente","foto_mes"]].equals(trlab[["numero_de_cliente","foto_mes"]]),
              "Final train keys desalineadas")
    train_ids=trdf.numero_de_cliente.to_numpy(np.int64)
    y=trlab.clase_ternaria.eq("BAJA+2").to_numpy(np.int8)
    months=trdf.foto_mes.to_numpy(np.int64)
    pc=positive_cluster_vector(train_ids,y,cmap)
    base=base_weight_vector(months,V1.FINAL_RECENCY[src["policy"]])
    weights,wd=A.make_weights(y,pc,base,winner)
    keep=weights>0
    x=trdf[features] if bool(np.all(keep)) else trdf.loc[keep,features].reset_index(drop=True)
    yfit=y[keep]; wfit=weights[keep]

    scdf,_=cargar_bloques(bank,meses=[202108],incluir_raw=True,features=derived)
    ids=scdf.numero_de_cliente.to_numpy(np.int64)
    A.require(len(ids)==164647 and len(np.unique(ids))==len(ids),"IDs agosto invalidos")
    xscore=scdf[features]

    identity=A.fingerprint({
        "run":RUN,"stage":4,"plan":PLAN,"confirm20_identity":s3["identity"],"code_sha":CODE_SHA,
        "winner":winner,"source":SOURCE_IDENTITY,"features":A.fingerprint(features),
        "params":src["params"],"rounds":src["rounds"],"policy":src["policy"],
        "weight_detail":wd,"n":A.N_FIXED,
    })
# [Anotación · líneas 510–541] Para cada una de las 20 semillas reutiliza el par modelo y `.npz` válido o
#   entrena un LightGBM con dataset propio, los parámetros y rondas de SUPERFINAL y los pesos por fila; puntúa
#   202108 y guarda el modelo en texto y el `.npz` con su hash y el detalle de pesos.
    modeldir=root/"models"; modeldir.mkdir(parents=True,exist_ok=True)
    scores=[]
    for i,seed in enumerate(SEEDS20,1):
        cp=modeldir/f"seed_{seed}.npz"; mp=modeldir/f"modelo_{seed}.txt"
        if cp.is_file() and mp.is_file():
            with np.load(cp,allow_pickle=False) as z:
                A.require(str(z["identity"].item())==identity and int(z["seed"].item())==seed,
                          f"Final CP metadata seed={seed}")
                A.require(np.array_equal(z["ids"],ids),f"Final CP IDs seed={seed}")
                A.require(str(z["model_sha256"].item())==A.digest(mp),f"Final model hash seed={seed}")
                sc=np.asarray(z["scores"],dtype=np.float64)
            print(f"REUSAR FINAL {i}/20 seed={seed}",flush=True)
        else:
            A.require(not cp.exists() and not mp.exists(),f"Par final parcial seed={seed}")
            print(f"ENTRENAR FINAL {i}/20 seed={seed}",flush=True)
            ds,seed_params=V2.dataset_for_seed(x,yfit,wfit,features,src["params"],seed)
            model=lgb.train(seed_params,ds,num_boost_round=int(src["rounds"]))
            sc=np.asarray(model.predict(xscore,num_threads=C.THREADS),dtype=np.float64)
            A.require(sc.shape==ids.shape and np.isfinite(sc).all(),"Scores final invalidos")
            tmpm=mp.with_suffix(".tmp")
            model.save_model(str(tmpm)); tmpm.replace(mp)
            tmp=cp.with_suffix(".tmp")
            with tmp.open("wb") as fh:
                np.savez_compressed(
                    fh,identity=np.array(identity),arm=np.array(winner),seed=np.array(seed),
                    ids=ids,scores=sc,model_sha256=np.array(A.digest(mp)),
                    weight_detail=np.array(json.dumps(wd,sort_keys=True))
                )
            tmp.replace(cp)
            del model,ds
            gc.collect()
        scores.append(sc.copy())
