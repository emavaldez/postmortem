def _gcs(cmd, check=True, **kw):
    """Respaldo opcional a Cloud Storage. Se omite cuando el bucket es el placeholder de publicacion."""
    if any("REEMPLAZAR_CON_BUCKET_PROPIO" in str(a) or "BUCKET_PRIVADO_NO_PUBLICADO" in str(a) for a in cmd):
        return None
    return subprocess.run(cmd, check=check, **kw)

# [Anotación · líneas 69–95] `read_scores_dir` lee los puntajes guardados de una lista de semillas validando
#   identidad, etiqueta y claves. `control_gain_at_ns` calcula la ganancia promedio del promedio de semillas del
#   control en una lista de cortes.
def read_scores_dir(
    root: Path, identity: str, label: str, seeds: list[int]
) -> tuple[np.ndarray,np.ndarray,dict[int,np.ndarray]]:
    ids0=None;y0=None;out={}
    for seed in seeds:
        cp=root/f"seed_{seed}.npz"
        require(cp.is_file(),f"Falta checkpoint {cp}")
        with np.load(cp,allow_pickle=False) as z:
            require(str(z["identity"].item())==identity,f"Identity distinta {cp}")
            require(str(z["label"].item())==label,f"Label distinto {cp}")
            require(int(z["seed"].item())==seed,f"Seed distinta {cp}")
            ids=np.asarray(z["ids"],dtype=np.int64)
            y=np.asarray(z["y"],dtype=np.int8)
            sc=np.asarray(z["scores"],dtype=np.float64)
        if ids0 is None:
            ids0=ids.copy();y0=y.copy()
        else:
            require(np.array_equal(ids0,ids) and np.array_equal(y0,y),"Checkpoints desalineados")
        require(sc.shape==ids.shape and np.isfinite(sc).all(),f"Scores inválidos {cp}")
        out[seed]=sc.copy()
    return ids0,y0,out


def control_gain_at_ns(ids: np.ndarray,y: np.ndarray,scores: dict[int,np.ndarray],seeds:list[int],ns:list[int]) -> float:
    mat=np.vstack([scores[s] for s in seeds])
    curve=gain_curve(ids,y,mat.mean(axis=0,dtype=np.float64))
    return float(np.mean([curve[n] for n in ns]))


# [Anotación · líneas 98–110] `run_b1_if_needed` ejecuta `run_b1.py --prepare` y `--execute` como subprocesos

    require(b1.is_file(),f"Falta runner B1: {b1}")
    py=Path(sys.executable)
    print("STAGE0: B1 prepare",flush=True)
    subprocess.run([str(py),str(b1),"--prepare","--home",str(Path.home())],check=True)
    print("STAGE0: B1 execute",flush=True)
    subprocess.run([str(py),str(b1),"--execute","--home",str(Path.home())],check=True)

# [Anotación · líneas 398–431] `tune_objective_factory` carga entrenamiento 202103 y evaluación 202105 y
#   define el objetivo de Optuna: propone learning_rate (0,015–0,06, log), num_leaves (24–128), min_data_in_leaf
#   (100–1.200, log), feature_fraction (0,55–1), bagging_fraction (0,65–1), lambda_l1 (1e-8–5, log), lambda_l2
#   (1e-8–10, log), max_bin (63, 127 o 255) y rondas (200–550); entrena con semilla 17 y devuelve la ganancia
#   promedio de la meseta elegida en 202105.
def tune_objective_factory(ctx:dict,features:list[str],eval_bank:Path,out:Path):
    import lightgbm as lgb
    data=load_fold(ctx,features,TUNE_TRAIN,TUNE_EVAL,eval_bank)
    xtr=data["x_train"]; ytr=data["y_train"]; xev=data["x_eval"]
    ids=data["ids"];yev=data["y_eval"]

    def objective(trial):
        params=base_params(ctx["paths"])
        params.update({
            "learning_rate":trial.suggest_float("learning_rate",0.015,0.06,log=True),
            "num_leaves":trial.suggest_int("num_leaves",24,128),
            "min_data_in_leaf":trial.suggest_int("min_data_in_leaf",100,1200,log=True),
            "feature_fraction":trial.suggest_float("feature_fraction",0.55,1.0),
            "bagging_fraction":trial.suggest_float("bagging_fraction",0.65,1.0),
            "lambda_l1":trial.suggest_float("lambda_l1",1e-8,5.0,log=True),
            "lambda_l2":trial.suggest_float("lambda_l2",1e-8,10.0,log=True),
            "max_bin":trial.suggest_categorical("max_bin",[63,127,255]),
        })
        rounds=trial.suggest_int("rounds",200,550)
        ds=lgb.Dataset(
            xtr,label=ytr,feature_name=features,free_raw_data=True,
            params={"max_bin":int(params["max_bin"])}
        )
        model=lgb.train(dict(params,seed=17),ds,num_boost_round=int(rounds))
        scores=np.asarray(model.predict(xev,num_threads=THREADS),dtype=np.float64)
        curve=gain_curve(ids,yev,scores)
        choice=robust_choice(curve)
        trial.set_user_attr("n_robusto",choice["n_robusto"])
        trial.set_user_attr("gain_peak",choice["gain_peak"])
        del model,ds,scores
        gc.collect()
        return float(choice["ganancia_robusta"])

    return data,objective

        "stage":3,"stage2":s2["identity"],"features":fingerprint(features)
    })[:16]
    study=optuna.create_study(
        direction="maximize",
        sampler=optuna.samplers.TPESampler(seed=17),
        storage=storage,study_name=study_name,load_if_exists=True,
    )
    complete=[t for t in study.trials if t.state==optuna.trial.TrialState.COMPLETE]
    todo=max(0,20-len(complete))
    if todo:
        study.optimize(objective,n_trials=todo,gc_after_trial=True)
    complete=[t for t in study.trials if t.state==optuna.trial.TrialState.COMPLETE]
    require(len(complete)>=20,"Optuna no completó 20 trials")
    ranked=sorted(complete,key=lambda t:(t.value,-t.number),reverse=True)
    top=ranked[:3]

    trial_rows=[]
    for t in complete:
        trial_rows.append({"trial":t.number,"value":t.value,**t.params})
    pd.DataFrame(trial_rows).sort_values("value",ascending=False).to_csv(out/"trials.csv",index=False)

# [Anotación · líneas 467–545] `stage3`, revalidación: arma cuatro candidatos (BASE con parámetros base y 300
#   rondas, y los tres trials), los entrena con 202103–202104 y los pesos de la política ganadora, con 5
#   semillas, y los compara en 202106 contra BASE con la regla de meseta y un mínimo de 3 semillas a favor. Gana
#   el promovido de mayor ganancia de meseta, o BASE. `RESULT.json` guarda parámetros, rondas y número de trial.
    base=base_params(ctx["paths"])
    candidates=[{"name":"BASE","params":base,"rounds":300,"trial":None}]
    for t in top:
        p=base_params(ctx["paths"])
        tp=dict(t.params)
        rounds=int(tp.pop("rounds"))
        p.update(tp)
        candidates.append({"name":f"TUNE_{t.number}","params":p,"rounds":rounds,"trial":t.number})

    data=load_fold(ctx,features,TRAIN_EVAL,EVAL_MONTH,eval_bank)
    wmap=RECENCY[s2["winner_policy"]]
    scores_by={}
    meta={}
    base_id=fingerprint({
        "stage":3,"plan":plan_doc(),"stage2":s2["identity"],
        "winner_policy":s2["winner_policy"],"features":fingerprint(features)
    })
    for cand in candidates:
        cid=fingerprint({"parent":base_id,"candidate":cand})
        sc=train_seeds(
            data,features,cand["params"],cand["rounds"],SEEDS5,wmap,
            out/"confirm"/cand["name"],cid,cand["name"]
        )
        choice,curves,ens=summarize_scores(data["ids"],data["y_eval"],sc,SEEDS5)
        scores_by[cand["name"]]={k:v.copy() for k,v in sc.items()}
        meta[cand["name"]]={"identity":cid,"choice":choice,"params":cand["params"],"rounds":cand["rounds"],"trial":cand["trial"]}
        if cand["name"]=="BASE": control_curves=curves
        del sc,curves,ens
        gc.collect()

    base_scores=scores_by["BASE"]
    rows=[];promoted=[]
    for cand in candidates:
        name=cand["name"];choice=meta[name]["choice"]
        if name=="BASE":
            paired={"mean":0.0,"median":0.0,"sd":0.0,"positive_fraction":0.0,"wins":0,"ties":5,"losses":0}
            c_same=choice["ganancia_robusta"];passed=True
        else:
            _,curves,_=summarize_scores(data["ids"],data["y_eval"],scores_by[name],SEEDS5)
            paired=paired_at_plateau(curves,control_curves,SEEDS5,choice["meseta_n"])
            c_same=control_gain_at_ns(data["ids"],data["y_eval"],base_scores,SEEDS5,choice["meseta_n"])
            passed=bool(
                choice["ganancia_robusta"]>c_same and paired["mean"]>0 and
                paired["median"]>0 and paired["wins"]>=3
            )
            if passed: promoted.append(name)
        rows.append({
            "candidate":name,"trial":cand["trial"],"rounds":cand["rounds"],"passed":passed,
            "control_gain_same_plateau":float(c_same),
            "delta_ensemble_same_plateau":float(choice["ganancia_robusta"]-c_same),
            **{k:v for k,v in choice.items() if k!="meseta_gain"},
            **{f"paired_{k}":v for k,v in paired.items() if k!="deltas"},
        })
    df=pd.DataFrame(rows);df.to_csv(out/"confirmation.csv",index=False)
    if promoted:
        winner=(
            df[df.candidate.isin(promoted)]
            .sort_values(["ganancia_robusta","delta_ensemble_same_plateau"],ascending=[False,False])
            .iloc[0].candidate
        )
    else:
        winner="BASE"
    m=meta[winner]
    r={
        "status":"completed","identity":base_id,"winner":winner,
        "params":m["params"],"rounds":m["rounds"],"trial":m["trial"],
        "winner_policy":s2["winner_policy"],"representation":s1["winner"],
        "quality_arm":s0["quality_arm"],"winner_features_file":s1["winner_features_file"],
        "promoted":promoted,
    }
    write_json(result_path,r)
    _gcs(
        ["gcloud","storage","cp",str(result_path),str(out/"trials.csv"),str(out/"confirmation.csv"),REMOTE+"/"],
        check=True
    )
    print("STAGE3 COMPLETO. Params:",winner,flush=True)
    del data,data_may
    gc.collect()
    return r


# [Anotación · líneas 548–612] `stage4`: entrena el candidato (variables, política, parámetros y rondas
#   elegidos) con las 20 semillas sobre 202103–202104, reutilizando las 5 de la etapa 3, y evalúa en 202106. El
#   control son los 20 puntajes de C3 del paso 2. Guarda la meseta del candidato, la comparación pareada,
#   `n_final` (corte central de la meseta) y `promote_to_august` con un mínimo de 12 semillas a favor, que la
#   versión 2 recalcula con 14.
def stage4(ctx:dict,s0:dict,s1:dict,s2:dict,s3:dict)->dict:
    out=stage_path(ctx,4,"confirm20")
    result_path=out/"RESULT.json"
    if result_path.is_file():
        return load_json(result_path)

    features=load_json(Path(s1["winner_features_file"]))["features"]
    eval_bank=Path(s0["eval_bank"])
    data=load_fold(ctx,features,TRAIN_EVAL,EVAL_MONTH,eval_bank)
    wmap=RECENCY[s2["winner_policy"]]
    identity=fingerprint({
        "stage":4,"plan":plan_doc(),"stage3":s3["identity"],
        "features":fingerprint(features),"params":s3["params"],
        "rounds":s3["rounds"],"policy":s2["winner_policy"]
    })

    # Reutilizar las primeras 5 del candidato elegido en stage3.
    s3out=stage_path(ctx,3,"tuning")
    cdir=s3out/"confirm"/s3["winner"]
    cp0=cdir/f"seed_{SEEDS5[0]}.npz"
    with np.load(cp0,allow_pickle=False) as z:
        oldid=str(z["identity"].item());oldlabel=str(z["label"].item())
    _,_,reuse=read_scores_dir(cdir,oldid,oldlabel,SEEDS5)

    scores=train_seeds(
        data,features,s3["params"],int(s3["rounds"]),SEEDS20,wmap,
        out/"candidate",identity,"SUPERFINAL",reuse_scores=reuse
    )
    choice,cand_curves,ens=summarize_scores(data["ids"],data["y_eval"],scores,SEEDS20)

    ids_c,y_c,control=load_s3_control(ctx,SEEDS20)
    require(np.array_equal(ids_c,data["ids"]) and np.array_equal(y_c,data["y_eval"]),
            "Control S3 y candidato desalineados")
    control_curves={s:gain_curve(ids_c,y_c,control[s]) for s in SEEDS20}
    paired=paired_at_plateau(cand_curves,control_curves,SEEDS20,choice["meseta_n"])
    c_same=control_gain_at_ns(ids_c,y_c,control,SEEDS20,choice["meseta_n"])
    promote=bool(
        choice["ganancia_robusta"]>c_same and paired["mean"]>0 and
        paired["median"]>0 and paired["wins"]>=12
    )
    cmat=np.vstack([control[s] for s in SEEDS20])
    ccurve=gain_curve(ids_c,y_c,cmat.mean(axis=0,dtype=np.float64))
    cchoice=robust_choice(ccurve)

    r={
        "status":"completed","identity":identity,"promote_to_august":promote,
        "candidate_choice":choice,"control_c3_choice":cchoice,
        "control_gain_at_candidate_plateau":float(c_same),
        "delta_ensemble_at_candidate_plateau":float(choice["ganancia_robusta"]-c_same),
        "paired":{k:v for k,v in paired.items() if k!="deltas"},
        "n_final":int(choice["n_robusto"]),
        "representation":s1["winner"],"quality_arm":s0["quality_arm"],
        "winner_policy":s2["winner_policy"],"params":s3["params"],
        "rounds":int(s3["rounds"]),"winner_features_file":s1["winner_features_file"],
    }
    write_json(result_path,r)
    pd.DataFrame([
        {"seed":s,"delta_meseta":paired["deltas"][i]}
        for i,s in enumerate(SEEDS20)
    ]).to_csv(out/"deltas.csv",index=False)
    _gcs(["gcloud","storage","cp",str(result_path),str(out/"deltas.csv"),REMOTE+"/"],check=True)
    print("STAGE4 COMPLETO. PROMOTE=",promote,"N=",r["n_final"],flush=True)
    del data,scores,ens,control
    gc.collect()
    return r


# [Anotación · líneas 615–649] `final_blocks` obtiene del catálogo los bloques de las variables elegidas.
#   `prepare_final_bank` genera (o valida) el banco final con esos bloques sobre el dataset de la etapa 0, con
#   meses de salida 202103–202106 y 202108, y verifica que las definiciones coincidan con el banco original.
def final_blocks(ctx:dict,features:list[str])->list[str]:
    cat=pd.read_csv(ctx["paths"]["bank_orig"]/"catalogo_features.csv").set_index("feature",verify_integrity=True)
    raw=set(ctx["schema"][:152])
    derived=[f for f in features if f not in raw]
    require(set(derived)<=set(cat.index),"Features final fuera de catálogo")
    return sorted(cat.loc[derived,"block"].dropna().unique().tolist())


def prepare_final_bank(ctx:dict,s0:dict,features:list[str])->Path:
    p=ctx["paths"]
    bank=p["out"]/"stage5_final"/"bank"
    data=Path(s0["dataset"])
    blocks=final_blocks(ctx,features)
    config=load_json(p["stage"]/"config_screening.json")
    months=FINAL_TRAIN+[SCORE_MONTH]
    if (bank/"manifest.json").is_file():
        m=load_json(bank/"manifest.json")
        require(m.get("status")=="completed","Final bank incompleto")
        require(m.get("input_sha256")==digest(data),"Final bank input distinto")
        require(m.get("output_months")==months,"Final bank meses distintos")
        require(m.get("blocks_requested")==blocks,"Final bank bloques distintos")
        validate_catalog_compat(p["bank_orig"],bank,features)
        return bank
    if bank.exists() and any(bank.iterdir()):
        q=bank.parent/("bank_incomplete_"+str(int(time.time())))
        bank.rename(q)
    add_fe_path(p)
    from generar_fe import generate
    print("STAGE5: generar banco final",blocks,flush=True)
    generate(data,bank,config,blocks=blocks,months=months)
    m=load_json(bank/"manifest.json")
    require(m.get("status")=="completed" and m.get("input_sha256")==digest(data),
            "Final bank no cerró")
    validate_catalog_compat(p["bank_orig"],bank,features)
    return bank


# [Anotación · líneas 652–768] `stage5` de la versión 1: entrenamiento final con un `lgb.Dataset` compartido y
#   exportación con prefijo `C1SF`. La versión 2 no la llama; usa `stage5_per_seed_safe`.
def stage5(ctx:dict,s0:dict,s1:dict,s2:dict,s3:dict,s4:dict)->dict:
    import lightgbm as lgb

    out=stage_path(ctx,5,"final")
    result_path=out/"RESULT.json"
    if result_path.is_file():
        return load_json(result_path)
    if not s4["promote_to_august"]:
        r={"status":"stopped_no_promotion","stage4_identity":s4["identity"],"no_submit":True}
        write_json(result_path,r)
        _gcs(["gcloud","storage","cp",str(result_path),REMOTE+"/stage5_RESULT.json"],check=True)
        print("STAGE5 NO EJECUTADO: stage4 no promovió.",flush=True)
        return r

    features=load_json(Path(s1["winner_features_file"]))["features"]
    bank=prepare_final_bank(ctx,s0,features)
    add_fe_path(ctx["paths"])
    from cargar_bloques import cargar_bloques
    derived=split_derived(features,ctx)

    trdf,_=cargar_bloques(bank,meses=FINAL_TRAIN,incluir_raw=True,features=derived)
    labels=pd.read_parquet(ctx["paths"]["bank_orig"]/"etiquetas.parquet")
    trlab=labels[labels.foto_mes.isin(FINAL_TRAIN)].reset_index(drop=True)
    require(trdf[["numero_de_cliente","foto_mes"]].equals(trlab[["numero_de_cliente","foto_mes"]]),
            "Final train labels desalineadas")
    xtr=trdf[features]
    ytr=trlab.clase_ternaria.eq("BAJA+2").to_numpy(np.int8)
    months=trdf.foto_mes.to_numpy(np.int64)
    keep,w=policy_weights(months,FINAL_RECENCY[s2["winner_policy"]])
    xtr2=xtr.loc[keep].reset_index(drop=True)
    ytr2=ytr[keep]
    ds=lgb.Dataset(
        xtr2,label=ytr2,weight=w,feature_name=features,free_raw_data=True,
        params={"max_bin": int(s3["params"].get("max_bin",255))}
    )
    ds.construct()
    del trdf,trlab,xtr,xtr2,ytr,ytr2,months,keep,w,labels
    gc.collect()

    scdf,_=cargar_bloques(bank,meses=[SCORE_MONTH],incluir_raw=True,features=derived)
    require(len(scdf)==EXPECTED_ROWS[SCORE_MONTH],"Filas agosto inesperadas")

    require(len(np.unique(ids))==len(ids),"IDs agosto duplicados")

    gc.collect()

    identity=fingerprint({
        "stage":5,"plan":plan_doc(),"stage4":s4["identity"],
        "data_sha":digest(Path(s0["dataset"])),"bank_manifest_sha":digest(bank/"manifest.json"),
        "features":fingerprint(features),"params":s3["params"],"rounds":s3["rounds"],
        "policy":s2["winner_policy"],"n":s4["n_final"]
    })
    modeldir=out/"models";modeldir.mkdir(exist_ok=True)
    all_scores=[]
    for i,seed in enumerate(SEEDS20,1):
        cp=modeldir/f"seed_{seed}.npz";mp=modeldir/f"modelo_{seed}.txt"

                require(np.array_equal(z["ids"],ids),f"Final IDs distintos {seed}")

            scores=np.asarray(model.predict(xscore,num_threads=THREADS),dtype=np.float64)
            require(scores.shape==ids.shape and np.isfinite(scores).all(),"Scores final inválidos")
            tmpm=mp.with_suffix(".tmp");model.save_model(str(tmpm));tmpm.replace(mp)
            with cp.with_suffix(".tmp").open("wb") as fh:
                np.savez_compressed(
                    fh,identity=np.array(identity),seed=np.array(seed),ids=ids,scores=scores,
                    model_sha256=np.array(digest(mp))
                )
            cp.with_suffix(".tmp").replace(cp)
            _gcs(["gcloud","storage","cp",str(cp),str(mp),REMOTE+"/stage5_models/"],check=True)

            gc.collect()
        all_scores.append(scores)

    scoremat=np.vstack(all_scores)

# [Anotación · líneas 771–825] `run_all` y `main` de la versión 1 (`--all` o `--stage N`). El camino de la

def run_all(ctx:dict)->dict:
    s0=stage0(ctx)
