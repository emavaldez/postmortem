# [Anotación · líneas 80–93] Define primos_entre, una criba de Eratóstenes que devuelve los primos de un
#   rango. Fija el generador con la semilla primigenia (Mersenne-Twister, muestreo por rechazo) y sortea 50
#   primos entre 100.000 y 1.000.000. De esa lista, las primeras 30 semillas son para el ensamble final y las
#   primeras 5 para la curva de 202106.
# Semillas del ensamble final: salen todas de la semilla primigenia, con el método de la cátedra
# (se sortean números primos entre 100.000 y 1.000.000). Con la misma primigenia, la lista es siempre la misma.
primos_entre <- function(desde, hasta) {          # criba de Eratóstenes
  es_primo <- rep(TRUE, hasta)
  es_primo[1] <- FALSE
  for (i in 2:floor(sqrt(hasta))) if (es_primo[i]) es_primo[seq(i * i, hasta, by= i)] <- FALSE
  primos <- which(es_primo)
  primos[primos >= desde]
}

# [Anotación · líneas 145–177] Tres funciones. ranking_cero_fijo divide el rango de cada valor positivo
#   (empates promediados) por la cantidad de positivos, hace lo mismo con signo negativo para los negativos, y
#   deja el cero en 0 y los NA en NA. calcular_auc obtiene el AUC a partir de la suma de rangos de los casos
#   positivos. curva_ganancia ordena por probabilidad descendente y acumula 1.072.500 por cada BAJA+2 y -27.500
#   por cada otro cliente, en millones.
# %%
# ------------------------------------------------------------------------------
# 2. FUNCIONES AUXILIARES
# ------------------------------------------------------------------------------

# Ranking con cero fijo de un vector (una variable, en un mes):
#   positivos -> (0, 1]    negativos -> [-1, 0)    cero -> 0    NA -> NA
ranking_cero_fijo <- function(x) {
  x   <- as.numeric(x)
  res <- x                      # así los 0 quedan en 0 y los NA en NA
  pos <- which(x > 0)
  neg <- which(x < 0)
  # Positivos: proporción de positivos del mes con valor <= x  (el mayor = 1)
  if (length(pos) > 0) res[pos] <-  frank( x[pos], ties.method = "average") / length(pos)
  # Negativos: lo mismo sobre el valor absoluto, con signo menos (el más negativo = -1)
  if (length(neg) > 0) res[neg] <- -frank(-x[neg], ties.method = "average") / length(neg)
  res
}

# AUC a partir de las probabilidades y la clase 0/1
calcular_auc <- function(prob, clase01) {
  r  <- frank(prob, ties.method = "average")
  n1 <- sum(clase01 == 1L)
  n0 <- length(clase01) - n1
  (sum(r[clase01 == 1L]) - n1 * (n1 + 1) / 2) / (n1 * n0)
}

# Ganancia acumulada, en millones, enviando el estímulo de mayor a menor probabilidad.
# El elemento k del resultado es la ganancia de enviar a los k clientes con mayor probabilidad.
curva_ganancia <- function(prob, clase_ternaria) {
  gan <- ifelse(clase_ternaria == "BAJA+2", PARAM$ganancia_acierto, -PARAM$costo_estimulo)
  cumsum(gan[order(-prob)]) / 1e6
}

# [Anotación · líneas 310–390] Compila con Rcpp la función fhistC. Para cada fila recorre la ventana que va
#   desde la posición indicada en pdesde hasta la fila actual, junta los valores que no son NA con su posición
#   dentro de la ventana y, si hay al menos dos, calcula la pendiente de la recta de mínimos cuadrados, el
#   mínimo, el máximo y el promedio; si no, deja NA. Devuelve un vector de largo 5n con esos cuatro resultados y
#   el valor de la fila anterior. El comentario indica que es código publicado en Zulip, sin modificar.
# %%
# ------------------------------------------------------------------------------
# 3e. TENDENCIA Y PROMEDIO DE LOS ÚLTIMOS 3 MESES, CALCULADOS SOBRE LOS RANKINGS
# Las dos funciones que siguen son el código de Feature Engineering Histórico
# que publicó Gustavo en Zulip, sin modificar.
# Para cada variable original se agregan:
#   <variable>_tend3 : pendiente de la recta que ajusta los últimos 3 meses
#   <variable>_avg3  : promedio de los últimos 3 meses
# ------------------------------------------------------------------------------


cppFunction("NumericVector fhistC(NumericVector pcolumna, IntegerVector pdesde )
{
  /* Aqui se cargan los valores para la regresion */
  double  x[100] ;
  double  y[100] ;

  int n = pcolumna.size();
  NumericVector out( 5*n );

  for(int i = 0; i < n; i++)
  {
    //lag
    if( pdesde[i]-1 < i )  out[ i + 4*n ]  =  pcolumna[i-1] ;
    else                   out[ i + 4*n ]  =  NA_REAL ;


    int  libre    = 0 ;
    int  xvalor   = 1 ;

    for( int j= pdesde[i]-1;  j<=i; j++ )
    {
       double a = pcolumna[j] ;

       if( !R_IsNA( a ) )
       {
          y[ libre ]= a ;
          x[ libre ]= xvalor ;
          libre++ ;
       }

       xvalor++ ;
    }

    /* Si hay al menos dos valores */
    if( libre > 1 )
    {
      double  xsum  = x[0] ;
      double  ysum  = y[0] ;
      double  xysum = xsum * ysum ;
      double  xxsum = xsum * xsum ;
      double  vmin  = y[0] ;
      double  vmax  = y[0] ;

      for( int h=1; h<libre; h++)
      {
        xsum  += x[h] ;
        ysum  += y[h] ;
        xysum += x[h]*y[h] ;
        xxsum += x[h]*x[h] ;

        if( y[h] < vmin )  vmin = y[h] ;
        if( y[h] > vmax )  vmax = y[h] ;
      }

      out[ i ]  =  (libre*xysum - xsum*ysum)/(libre*xxsum -xsum*xsum) ;
      out[ i + n ]    =  vmin ;
      out[ i + 2*n ]  =  vmax ;
      out[ i + 3*n ]  =  ysum / libre ;
    }
    else
    {
      out[ i       ]  =  NA_REAL ;
      out[ i + n   ]  =  NA_REAL ;
      out[ i + 2*n ]  =  NA_REAL ;
      out[ i + 3*n ]  =  NA_REAL ;
    }
  }

  return  out;
}")
# [Anotación · líneas 391–460] Define TendenciaYmuchomas. Construye vector_desde, que para cada fila indica
#   dónde empieza su ventana: la fila actual menos (ventana - 1), sin pasar de la primera fila del mismo
#   cliente. Para cada columna llama a fhistC y, según los argumentos, agrega al dataset las columnas _tend,
#   _min, _max, _avg, _ratioavg y _ratiomax con el tamaño de ventana como sufijo. El valor por defecto de
#   ventana es 6.
#------------------------------------------------------------------------------


TendenciaYmuchomas <- function(
    dataset, cols, ventana = 6, tendencia = TRUE,
    minimo = TRUE, maximo = TRUE, promedio = TRUE,
    ratioavg = FALSE, ratiomax = FALSE) {
  gc(verbose= FALSE)
  # Esta es la cantidad de meses que utilizo para la historia
  ventana_regresion <- ventana

  last <- nrow(dataset)

  # creo el vector_desde que indica cada ventana
  # de esta forma se acelera el procesamiento ya que lo hago una sola vez
  vector_ids <- dataset[ , numero_de_cliente ]

  vector_desde <- seq(
    -ventana_regresion + 2,
    nrow(dataset) - ventana_regresion + 1
  )

  vector_desde[1:ventana_regresion] <- 1

  for (i in 2:last) {
    if (vector_ids[i - 1] != vector_ids[i]) {
      vector_desde[i] <- i
    }
  }
  for (i in 2:last) {
    if (vector_desde[i] < vector_desde[i - 1]) {
      vector_desde[i] <- vector_desde[i - 1]
    }
  }

  for (campo in cols) {
    nueva_col <- fhistC(dataset[, get(campo)], vector_desde)

    if (tendencia) {
      dataset[, paste0(campo, "_tend", ventana) :=
        nueva_col[(0 * last + 1):(1 * last)]]
    }

    if (minimo) {
      dataset[, paste0(campo, "_min", ventana) :=
        nueva_col[(1 * last + 1):(2 * last)]]
    }

    if (maximo) {
      dataset[, paste0(campo, "_max", ventana) :=
        nueva_col[(2 * last + 1):(3 * last)]]
    }

    if (promedio) {
      dataset[, paste0(campo, "_avg", ventana) :=
        nueva_col[(3 * last + 1):(4 * last)]]
    }

    if (ratioavg) {
      dataset[, paste0(campo, "_ratioavg", ventana) :=
        get(campo) / nueva_col[(3 * last + 1):(4 * last)]]
    }

    if (ratiomax) {
      dataset[, paste0(campo, "_ratiomax", ventana) :=
        get(campo) / nueva_col[(2 * last + 1):(3 * last)]]
    }
  }
}

# [Anotación · líneas 520–548] Si BO_log.txt ya tiene al menos 12 filas, saltea la optimización; si está

#   10.500 y en 11.000 envíos, la ganancia máxima, la cantidad de envíos donde ocurre y los minutos. Devuelve el
#   AUC como valor a maximizar.
# Si la optimización ya está terminada en la carpeta de trabajo, no se repite
bo_hecha <- file.exists("BO_log.txt") && nrow(fread("BO_log.txt")) >= PARAM$hyperparametertuning$iteraciones
if (!bo_hecha) {
  if (file.exists("BO_log.txt")) file.remove("BO_log.txt")    # quedó a medias: se empieza de nuevo
  giter <- 0

  # Entrena un modelo con los hiperparámetros xs y devuelve su AUC en 202106
  Estimar_lightgbm <- function(xs) {
    flush.console()
    giter <<- giter + 1
    t0 <- Sys.time()

    param_completo <- modifyList(PARAM$lgbm$param_fijos, xs)
    modelo <- lgb.train(data= dtrain, param= param_completo)
    prob <- predict(modelo, mvalidate)
    ganancia <- curva_ganancia(prob, validate_clase)              # solo informativo, no se usa para elegir
    registro <- data.table(
      iter= giter, num_iterations= xs$num_iterations, num_leaves= xs$num_leaves,
      min_sum_hessian_in_leaf= xs$min_sum_hessian_in_leaf,
      metrica= calcular_auc(prob, validate_y),                    # lo que se optimiza

      ganancia_max= max(ganancia), envios_max= which.max(ganancia),
      minutos= round(as.numeric(difftime(Sys.time(), t0, units= "mins")), 1))
    fwrite(registro, file= "BO_log.txt", sep= "\t", append= file.exists("BO_log.txt"))
    print(registro)

    set.seed(PARAM$semilla_primigenia, kind = "L'Ecuyer-CMRG")
    return( list( metrica = registro$metrica) )
  }

# [Anotación · líneas 550–583] Configura mlr3mbo: función objetivo sobre el espacio de búsqueda con metrica a
#   maximizar, terminador de 12 evaluaciones, surrogate de kriging (regr.km con BFGS y nugget estimado), función
#   de adquisición Expected Improvement recorrida con búsqueda al azar de 1.000 puntos, y ciclo bayesopt_ego.
#   Fija la semilla primigenia con el generador L'Ecuyer-CMRG y ejecuta la optimización.
  cat("Iniciando Optimización Bayesiana...\n")
  objective <- ObjectiveRFun$new(
    fun= Estimar_lightgbm,
    domain= PARAM$hypeparametertuning$hs,
    codomain= ps(metrica = p_dbl(tags="maximize")),
    id= "BO_lightgbm"
  )

  instance <- OptimInstanceBatchSingleCrit$new(
    objective= objective,
    terminator=  trm("evals", n_evals = PARAM$hyperparametertuning$iteraciones)
  )

  surrogate <- srlrn(lrn("regr.km",
                         optim.method = "BFGS",
                         nugget.estim = TRUE,
                         control= list(trace = FALSE)))

  acq_function  <- acqf("ei")
  acq_optimizer  <- acqo(
    optimizer  = opt("random_search", batch_size= 1000),
    terminator = trm("evals", n_evals= 1000)
  )

  optimizer <- opt("mbo",
                   loop_function = bayesopt_ego,
                   surrogate= surrogate,
                   acq_function= acq_function,
                   acq_optimizer= acq_optimizer
  )

  set.seed(PARAM$semilla_primigenia, kind = "L'Ecuyer-CMRG")
  bayesiana_salida <- optimizer$optimize(instance)
}
