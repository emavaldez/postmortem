#%% celda 43 [código]
# particionar agrega una columna llamada fold a un dataset
#   que consiste en una particion estratificada segun agrupa
# particionar( data=dataset, division=c(70,30),
#  agrupa=clase_ternaria, seed=semilla)   crea una particion 70, 30

particionar <- function(data, division, agrupa= "", campo= "fold", start= 1, seed= NA) {
  if (!is.na(seed)) set.seed(seed, "L'Ecuyer-CMRG")

  bloque <- unlist(mapply(
    function(x, y) {rep(y, x)},division, seq(from= start, length.out= length(division))))

  data[, (campo) := sample(rep(bloque,ceiling(.N / length(bloque))))[1:.N],by= agrupa]
}

  return( drealidad )
}

realidad_evaluar <- function( prealidad, pprediccion) {

  prealidad[ pprediccion,
    on= c("numero_de_cliente", "foto_mes"),
    predicted:= i.Predicted
  ]

  tbl <- prealidad[, list("qty"=.N), list(fold, predicted, clase_ternaria)]

  res <- list()
  res$public  <- tbl[fold==1 & predicted==1L, sum(qty*ifelse(clase_ternaria=="BAJA+2", 1072500, -27500))]/0.3
  res$private <- tbl[fold==2 & predicted==1L, sum(qty*ifelse(clase_ternaria=="BAJA+2", 1072500, -27500))]/0.7
  res$total <- tbl[predicted==1L, sum(qty*ifelse(clase_ternaria=="BAJA+2", 1072500, -27500))]

  prealidad[, predicted:=NULL]
  return( res )
}

#%% celda 55 [código]
# En el argumento x llegan los parmaetros de la bayesiana
#  devuelve la AUC en cross validation del modelo entrenado

EstimarGanancia_AUC_lightgbm <- function(x) {

  # x pisa (o agrega) a param_fijos
  param_completo <- modifyList(PARAM$lgbm$param_fijos, x)

  # entreno LightGBM
  modelocv <- lgb.cv(
    data= dtrain,
    nfold= PARAM$hyperparametertuning$xval_folds,
    stratified= TRUE,
    param= param_completo
  )

  # obtengo la ganancia
  AUC <- modelocv$best_score

  # hago espacio en la memoria
  rm(modelocv)
  gc(full= TRUE, verbose= FALSE)

  message(format(Sys.time(), "%a %b %d %X %Y"), " AUC ", AUC)

  return(AUC)
}

## 2.4 Validación con semillas (ganador HT4942)

#%% celda 95 [código]
# leo el ganador de HT4942 (metrica average_precision, rango amplio)
tb_bayesiana_4942 <- fread(file.path(carpeta, "exp", "HT4942", "BO_log.txt"))
setorder(tb_bayesiana_4942, -y)
ganador_4942 <- tb_bayesiana_4942[1, list(num_iterations, learning_rate, feature_fraction, num_leaves, min_data_in_leaf)]
print(ganador_4942)

# undersampling real con el que se corrio HT4942 (no el actual de PARAM, que ahora es 0.3 por HT4944)
undersampling_4942 <- 0.1

# armo los hiperparametros finales con ese ganador
param_final_4942 <- modifyList(PARAM$lgbm$param_fijos, ganador_4942)
param_normalizado_4942 <- copy(param_final_4942)
param_normalizado_4942$min_data_in_leaf <- round(param_final_4942$min_data_in_leaf / undersampling_4942)

# genero 5 semillas frescas (mismas que ya usamos, para poder comparar contra la validacion de HT4940)
if (!require("primes")) install.packages("primes")
require("primes")
primos <- generate_primes(min = 100000, max = 1000000)
set.seed(PARAM$semilla_primigenia)
semillas_validacion <- sample(primos, 5)

drealidad <- realidad_inicializar(dfuture, PARAM)

resultados_semillas_4942 <- data.table(semilla = semillas_validacion, ganancia = NA_real_)

for (i in seq_along(semillas_validacion)) {
  s <- semillas_validacion[i]
  param_normalizado_4942$seed <- s

  modelo_val <- lgb.train(data = dtrain_final, param = param_normalizado_4942)
  prediccion_val <- predict(modelo_val, data.matrix(dfuture[, campos_buenos, with = FALSE]))

  tb_val <- dfuture[, list(numero_de_cliente, foto_mes, clase_ternaria)]
  tb_val[, prob := prediccion_val]
  setorder(tb_val, -prob)

  tb_val[, Predicted := 0L]
  tb_val[1:12500, Predicted := 1L] # corte fijo: el ganador de HT4942

  res <- realidad_evaluar(drealidad, tb_val)
  resultados_semillas_4942[i, ganancia := res$total]

  cat("Semilla", s, "-> Ganancia:", res$total, "\n")
}

cat("\nMedia:", resultados_semillas_4942[, mean(ganancia)], "\n")
cat("Desvio estandar:", resultados_semillas_4942[, sd(ganancia)], "\n")

fwrite(resultados_semillas_4942,
  file = file.path(carpeta, "exp", "HT4942", "validacion_5semillas.txt"),
  sep = "\t"
)
